cmake_minimum_required(VERSION 3.22)
project(SUBSTRATE VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# CUDA kernels (optional — skipped if nvcc not found)
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    add_subdirectory(cuda)
    message(STATUS "SUBSTRATE: CUDA kernels enabled (${CMAKE_CUDA_COMPILER})")
else()
    message(STATUS "SUBSTRATE: CUDA not found — skipping cuda/")
endif()
